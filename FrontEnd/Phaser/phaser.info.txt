// SocketContext.js (React side)
import { io } from "socket.io-client";
export const socket = io("http://localhost:3000", { autoConnect: false });

// Inside your React Component (e.g., GameContainer.jsx)
useEffect(() => {
  const config = {
    type: Phaser.AUTO,
    parent: 'phaser-game',
    scene: [MainGame],
    callbacks: {
      preBoot: (game) => {
        // Storing socket reference in Phaser's global registry
        game.registry.set('socket', socket);  >????????????????????????????   important
      }
    }
  };

  const game = new Phaser.Game(config);
  return () => game.destroy(true);
}, []);

// Inside your Phaser Scene class (e.g., MainGame.js)
update() {
  if (this.player.moved) {
    const socket = this.registry.get('socket');
    
    // Firing the event from the game to the backend
    socket.emit('player_movement', {
      x: this.player.x,
      y: this.player.y
    });
  }
}

// Inside your Phaser Scene's create() method
create() {
  const socket = this.registry.get('socket');

  // 1. Establish the network listener
  socket.on('enemy_moved', (data) => {
    // 2. Safely pass data into Phaser's internal event pipeline
    this.events.emit('network_enemy_update', data);
  });

  // 3. Listen to your own internal event safely within scene scope
  this.events.on('network_enemy_update', (data) => {
    // Correctly updates the Phaser sprite on the screen safely
    this.enemySprite.setPosition(data.x, data.y);
  });
}

important

To avoid memory leaks (where events duplicate every time a scene restarts), always clean up your listeners when the scene shuts down:

this.events.on('shutdown', () => {
  socket.off('enemy_moved');
});
